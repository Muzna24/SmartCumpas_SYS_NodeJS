CREATE TABLE Students (
    StudentID   NUMBER PRIMARY KEY,
    Name        VARCHAR2(100) NOT NULL,
    Email       VARCHAR2(100) NOT NULL UNIQUE,
    Password    VARCHAR2(255) DEFAULT 'password123'
);
/

CREATE TABLE Facilities (
    FacilityID    NUMBER,
    FacilityName  VARCHAR2(100) NOT NULL,
    PhotoPath     VARCHAR2(255),
    PRIMARY KEY (FacilityID)
);
/

CREATE TABLE Locations (
    LocationID    NUMBER PRIMARY KEY,
    BuildingName  VARCHAR2(100) NOT NULL,
    Floor         VARCHAR2(10),
    Description   VARCHAR2(255)
);
/

CREATE TABLE Time_Slot (
    TimeSlotID  NUMBER PRIMARY KEY,
    StartTime   DATE NOT NULL,
    EndTime     DATE NOT NULL
);
/

CREATE TABLE Booking (
    BookingID    NUMBER PRIMARY KEY,
    BookingDate  DATE NOT NULL,
    FacilityID   NUMBER,
    StudentID    NUMBER,
    TimeSlotID   NUMBER,
    CONSTRAINT FK_Booking_Student FOREIGN KEY (StudentID)
        REFERENCES Students (StudentID),
    CONSTRAINT FK_Booking_Facility FOREIGN KEY (FacilityID)
        REFERENCES Facilities(FacilityID),
    CONSTRAINT FK_Booking_TimeSlot FOREIGN KEY (TimeSlotID)
        REFERENCES Time_Slot(TimeSlotID)
);
/

CREATE TABLE Facility_Location (
    FacilityID  NUMBER,
    LocationID  NUMBER,
    Capacity    NUMBER CHECK (Capacity > 0),
    Status      VARCHAR2(20),
    PRIMARY KEY (FacilityID, LocationID),
    FOREIGN KEY (FacilityID) REFERENCES Facilities(FacilityID),
    FOREIGN KEY (LocationID) REFERENCES Locations(LocationID)
);
/

CREATE OR REPLACE FUNCTION Get_Max_Continuous_Availability (
    p_FacilityID  IN Facilities.FacilityID%TYPE,
    p_BookingDate IN DATE
) RETURN VARCHAR2
IS
    v_maxSlot        NUMBER;
    v_facility_name  Facilities.FacilityName%TYPE;
    v_val            NUMBER := 1;
    v_currentStart   NUMBER;
    v_bestStart      NUMBER;
    v_bestEnd        NUMBER;
    v_count          NUMBER;
    v_startTime      Time_Slot.StartTime%TYPE;
    v_endTime        Time_Slot.EndTime%TYPE;
BEGIN
    IF p_FacilityID IS NULL OR p_BookingDate IS NULL THEN
        RETURN 'Error: Invalid input (NULL values not allowed)';
    END IF;

    SELECT COUNT(*) INTO v_count FROM Facilities WHERE FacilityID = p_FacilityID;
    IF v_count = 0 THEN
        RETURN 'Error: Facility does not exist';
    END IF;

    SELECT MAX(TimeSlotID) INTO v_maxSlot FROM Time_Slot;
    SELECT FacilityName INTO v_facility_name FROM Facilities WHERE FacilityID = p_FacilityID;

    WHILE v_val <= v_maxSlot LOOP
        SELECT COUNT(*) INTO v_count
        FROM Booking
        WHERE FacilityID = p_FacilityID
          AND BookingDate = p_BookingDate
          AND TimeSlotID = v_val;

        IF v_count = 0 THEN
            IF v_currentStart IS NULL THEN
                v_currentStart := v_val;
            END IF;
        ELSE
            IF v_currentStart IS NOT NULL THEN
                v_bestStart := v_currentStart;
                v_bestEnd := v_val - 1;
                v_currentStart := NULL;
            END IF;
        END IF;

        v_val := v_val + 1;
    END LOOP;

    IF v_currentStart IS NOT NULL THEN
        v_bestStart := v_currentStart;
        v_bestEnd := v_maxSlot;
    END IF;

    IF v_bestStart IS NULL THEN
        RETURN 'No Availability in ' || v_facility_name;
    END IF;

    SELECT StartTime INTO v_startTime FROM Time_Slot WHERE TimeSlotID = v_bestStart;
    SELECT EndTime INTO v_endTime FROM Time_Slot WHERE TimeSlotID = v_bestEnd;

    RETURN 'The available time slots for facility: ' || v_facility_name || ' is from ' ||
           TO_CHAR(v_startTime, 'HH:MI AM') || ' - ' || TO_CHAR(v_endTime, 'HH:MI AM');
END Get_Max_Continuous_Availability;
/

CREATE OR REPLACE PROCEDURE GetAvailableTimeSlots (
    p_FacilityID  IN  Facilities.FacilityID%TYPE,
    p_BookingDate IN  DATE,
    p_Cursor      OUT SYS_REFCURSOR
) AS
    v_count NUMBER;
BEGIN
    IF p_FacilityID IS NULL OR p_BookingDate IS NULL THEN
        RAISE_APPLICATION_ERROR(-20001, 'Invalid input (NULL values not allowed)');
    END IF;

    SELECT COUNT(*) INTO v_count FROM Facilities WHERE FacilityID = p_FacilityID;
    IF v_count = 0 THEN
        RAISE_APPLICATION_ERROR(-20002, 'Facility does not exist');
    END IF;

    OPEN p_Cursor FOR
        SELECT
            ts.TimeSlotID                      AS "TimeSlotID",
            TO_CHAR(ts.StartTime, 'HH24:MI')   AS "StartTime",
            TO_CHAR(ts.EndTime, 'HH24:MI')     AS "EndTime"
        FROM Time_Slot ts
        WHERE ts.TimeSlotID NOT IN (
            SELECT TimeSlotID
            FROM Booking
            WHERE FacilityID = p_FacilityID
              AND BookingDate = p_BookingDate
        );
END GetAvailableTimeSlots;
/

CREATE OR REPLACE PROCEDURE GetAllSlotsWithStatus (
    p_FacilityID  IN  Facilities.FacilityID%TYPE,
    p_BookingDate IN  DATE,
    p_Cursor      OUT SYS_REFCURSOR
) AS
BEGIN
    OPEN p_Cursor FOR
        SELECT
            ts.TimeSlotID                    AS "TimeSlotID",
            TO_CHAR(ts.StartTime, 'HH24:MI') AS "StartTime",
            TO_CHAR(ts.EndTime, 'HH24:MI')   AS "EndTime",
            CASE WHEN b.TimeSlotID IS NULL THEN 'Available' ELSE 'Booked' END AS "Status"
        FROM Time_Slot ts
        LEFT JOIN Booking b
            ON b.TimeSlotID = ts.TimeSlotID
           AND b.FacilityID = p_FacilityID
           AND TRUNC(b.BookingDate) = TRUNC(p_BookingDate)
        ORDER BY ts.TimeSlotID;
END GetAllSlotsWithStatus;
/

CREATE OR REPLACE PROCEDURE AddBooking (
    p_BookingID   IN  Booking.BookingID%TYPE,
    p_BookingDate IN  Booking.BookingDate%TYPE,
    p_FacilityID  IN  Booking.FacilityID%TYPE,
    p_StudentID   IN  Booking.StudentID%TYPE,
    p_TimeSlotID  IN  Booking.TimeSlotID%TYPE,
    p_Result      OUT VARCHAR2
) AS
    v_count NUMBER;
BEGIN
    SELECT COUNT(*) INTO v_count
    FROM Facility_Location
    WHERE FacilityID = p_FacilityID AND Status = 'Inactive';

    IF v_count > 0 THEN
        p_Result := 'Error: Cannot book an inactive facility.';
        RETURN;
    END IF;

    SELECT COUNT(*) INTO v_count
    FROM Booking
    WHERE FacilityID = p_FacilityID
      AND TRUNC(BookingDate) = TRUNC(p_BookingDate)
      AND TimeSlotID = p_TimeSlotID;

    IF v_count > 0 THEN
        p_Result := 'Error: This time slot is already booked for this facility.';
        RETURN;
    END IF;

    INSERT INTO Booking (BookingID, BookingDate, FacilityID, StudentID, TimeSlotID)
    VALUES (p_BookingID, TRUNC(p_BookingDate), p_FacilityID, p_StudentID, p_TimeSlotID);

    p_Result := 'Booking added successfully.';
END AddBooking;
/

CREATE OR REPLACE PROCEDURE DeleteBooking (
    p_BookingID IN  Booking.BookingID%TYPE,
    p_Result    OUT VARCHAR2
) AS
    v_count NUMBER;
BEGIN
    SELECT COUNT(*) INTO v_count FROM Booking WHERE BookingID = p_BookingID;

    IF v_count = 0 THEN
        p_Result := 'Error: Booking not found.';
        RETURN;
    END IF;

    DELETE FROM Booking WHERE BookingID = p_BookingID;
    p_Result := 'Booking cancelled successfully.';
END DeleteBooking;
/

CREATE OR REPLACE PROCEDURE GetBookingsByStudent (
    p_StudentID IN  Booking.StudentID%TYPE,
    p_Cursor    OUT SYS_REFCURSOR
) AS
BEGIN
    OPEN p_Cursor FOR
        SELECT
            b.BookingID                          AS "BookingID",
            TO_CHAR(b.BookingDate, 'YYYY-MM-DD') AS "BookingDate",
            f.FacilityName                       AS "FacilityName",
            TO_CHAR(ts.StartTime, 'HH24:MI')     AS "StartTime",
            TO_CHAR(ts.EndTime, 'HH24:MI')       AS "EndTime",
            s.StudentID                          AS "StudentID",
            s.Name                                AS "StudentName"
        FROM Booking b
        JOIN Facilities f  ON b.FacilityID  = f.FacilityID
        JOIN Time_Slot ts  ON b.TimeSlotID  = ts.TimeSlotID
        JOIN Students s    ON b.StudentID   = s.StudentID
        WHERE b.StudentID = p_StudentID
        ORDER BY b.BookingDate, ts.StartTime;
END GetBookingsByStudent;
/

CREATE OR REPLACE PROCEDURE GetBusyReport (
    p_Cursor OUT SYS_REFCURSOR
) AS
BEGIN
    OPEN p_Cursor FOR
        SELECT
            f.FacilityID   AS "FacilityID",
            f.FacilityName AS "FacilityName",
            COUNT(b.BookingID) AS "BookingCount",
            CASE
                WHEN COUNT(b.BookingID) = 0 THEN 'Unused'
                WHEN COUNT(b.BookingID) > 5 THEN 'Busy'
                ELSE 'Not Busy'
            END AS "Status",
            ROUND(
                COUNT(b.BookingID) /
                NULLIF((SELECT COUNT(DISTINCT BookingDate) FROM Booking) * (SELECT COUNT(*) FROM Time_Slot), 0)
                * 100, 1
            ) AS "UtilizationPct",
            (
                SELECT TO_CHAR(ts.StartTime,'HH24:MI') || ' - ' || TO_CHAR(ts.EndTime,'HH24:MI')
                FROM Booking b2
                JOIN Time_Slot ts ON b2.TimeSlotID = ts.TimeSlotID
                WHERE b2.FacilityID = f.FacilityID
                GROUP BY ts.TimeSlotID, ts.StartTime, ts.EndTime
                ORDER BY COUNT(*) DESC
                FETCH FIRST 1 ROW ONLY
            ) AS "PeakHour"
        FROM Facilities f
        LEFT JOIN Booking b ON f.FacilityID = b.FacilityID
        GROUP BY f.FacilityID, f.FacilityName
        ORDER BY COUNT(b.BookingID) DESC;
END GetBusyReport;
/

CREATE OR REPLACE PROCEDURE AddFacility (
    p_FacilityName IN Facilities.FacilityName%TYPE,
    p_BuildingName IN Locations.BuildingName%TYPE,
    p_Floor        IN Locations.Floor%TYPE,
    p_Capacity     IN Facility_Location.Capacity%TYPE,
    p_Status       IN Facility_Location.Status%TYPE,
    p_PhotoPath    IN Facilities.PhotoPath%TYPE,
    p_Result       OUT VARCHAR2
) AS
    v_newFacilityID NUMBER;
    v_newLocationID NUMBER;
BEGIN
    SELECT NVL(MAX(FacilityID), 0) + 1 INTO v_newFacilityID FROM Facilities;
    SELECT NVL(MAX(LocationID), 0) + 1 INTO v_newLocationID FROM Locations;

    INSERT INTO Facilities (FacilityID, FacilityName, PhotoPath)
    VALUES (v_newFacilityID, p_FacilityName, p_PhotoPath);

    INSERT INTO Locations (LocationID, BuildingName, Floor, Description)
    VALUES (v_newLocationID, p_BuildingName, p_Floor, NULL);

    INSERT INTO Facility_Location (FacilityID, LocationID, Capacity, Status)
    VALUES (v_newFacilityID, v_newLocationID, p_Capacity, p_Status);

    p_Result := 'Facility added successfully.';
END AddFacility;
/

CREATE OR REPLACE PROCEDURE DeleteFacility (
    p_FacilityID IN Facilities.FacilityID%TYPE,
    p_Result     OUT VARCHAR2
) AS
    v_bookingCount NUMBER;
    v_locationId   NUMBER;
BEGIN
    SELECT COUNT(*) INTO v_bookingCount FROM Booking WHERE FacilityID = p_FacilityID;

    IF v_bookingCount > 0 THEN
        p_Result := 'Error: Cannot delete a facility that has existing bookings.';
        RETURN;
    END IF;

    SELECT LocationID INTO v_locationId FROM Facility_Location WHERE FacilityID = p_FacilityID;

    DELETE FROM Facility_Location WHERE FacilityID = p_FacilityID;
    DELETE FROM Facilities WHERE FacilityID = p_FacilityID;
    DELETE FROM Locations WHERE LocationID = v_locationId;

    p_Result := 'Facility deleted successfully.';
END DeleteFacility;
/

CREATE OR REPLACE PROCEDURE UpdateFacilityStatus (
    p_FacilityID IN  Facility_Location.FacilityID%TYPE,
    p_Status     IN  Facility_Location.Status%TYPE,
    p_Result     OUT VARCHAR2
) AS
BEGIN
    UPDATE Facility_Location SET Status = p_Status WHERE FacilityID = p_FacilityID;
    IF SQL%ROWCOUNT = 0 THEN
        p_Result := 'Error: Facility not found.';
    ELSE
        p_Result := 'Status updated successfully.';
    END IF;
END UpdateFacilityStatus;
/

CREATE OR REPLACE PROCEDURE RegisterStudent (
    p_Name       IN  Students.Name%TYPE,
    p_Email      IN  Students.Email%TYPE,
    p_Password   IN  VARCHAR2,
    p_NewID      OUT NUMBER,
    p_Result     OUT VARCHAR2
) AS
    v_emailCount NUMBER;
BEGIN
    SELECT COUNT(*) INTO v_emailCount FROM Students WHERE Email = p_Email;
    IF v_emailCount > 0 THEN
        p_Result := 'Error: This email is already registered.';
        RETURN;
    END IF;

    SELECT NVL(MAX(StudentID), 0) + 1 INTO p_NewID FROM Students;

    INSERT INTO Students (StudentID, Name, Email, Password)
    VALUES (p_NewID, p_Name, p_Email, p_Password);

    p_Result := 'Student registered successfully.';
END RegisterStudent;
/
